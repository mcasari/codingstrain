spring-kafka-demo
// ✅ Producer — send with KafkaTemplate
@Service
class MessageProducer {
    private final KafkaTemplate<String, String> kafka;

    MessageProducer(KafkaTemplate<String, String> kafka) {
        this.kafka = kafka;
    }

    void send(String message) {
        kafka.send("demo-topic", message);
    }
}
// ✅ Consumer — method becomes a listener
@Service
class MessageConsumer {
    @KafkaListener(topics = "demo-topic", groupId = "demo-group")
    void listen(String message) {
        System.out.println("Received: " + message);
    }
}

// ❌ Always load — blows up if Redis isn’t on the classpath
@Configuration
class RedisAutoConfig {
    @Bean
    RedisTemplate<String, String> redisTemplate() {
        return new RedisTemplate<>();
    }
}
// ✅ Only when Redis classes exist
@Configuration
@ConditionalOnClass(RedisTemplate.class)
class RedisAutoConfig {
    @Bean
    @ConditionalOnMissingBean
    RedisTemplate<String, String> redisTemplate() {
        return new RedisTemplate<>();
    }
}

# ❌ Lazy=true — @PostConstruct waits until first use
spring:
  main:
    lazy-initialization: true

@Service
class CacheWarmer {
    @PostConstruct
    void warm() {
        cache.loadAll(); // NOT at startup when lazy
    }
}
# ✅ Need init at boot? Stay eager (default) or force the bean
spring:
  main:
    lazy-initialization: false   # default — PostConstruct at startup

# Or keep lazy, but touch critical beans early:
@Bean
ApplicationRunner touch(CacheWarmer warmer) {
    return args -> { /* injecting warmer creates it now */ };
}
