import java.util.ArrayList;
import java.util.List;

// ❌ Inheritance — you get ArrayList's full API (add, remove, sort, …)
class LoggingList extends ArrayList<String> {
    @Override
    public boolean add(String item) {
        System.out.println("add: " + item);
        return super.add(item);   // coupled to ArrayList
    }
}

// ✅ Delegation — expose only what your class needs
class LoggingList {
    private final List<String> items = new ArrayList<>();

    public boolean add(String item) {
        System.out.println("add: " + item);
        return items.add(item);   // swap List impl anytime
    }

    public List<String> snapshot() {
        return List.copyOf(items);
    }
}

// ❌ Static cache grows forever — GC cannot reclaim entries
class SessionCache {
    private static final Map<String, byte[]> DATA = new HashMap<>();

    static void put(String id, byte[] payload) {
        DATA.put(id, payload); // never removed
    }
}
// ✅ Evict or remove when done — keep the map bounded
class SessionCache {
    private static final Map<String, byte[]> DATA = new HashMap<>();

    static void put(String id, byte[] payload) {
        DATA.put(id, payload);
    }

    static void remove(String id) {
        DATA.remove(id); // reachable → eligible for GC
    }
}

// ❌ Manual recursion with File — verbose and easy to miss dirs
void findJava(File dir, List<File> out) {
    File[] kids = dir.listFiles();
    if (kids == null) return;
    for (File f : kids) {
        if (f.isDirectory()) findJava(f, out);
        else if (f.getName().endsWith(".java")) out.add(f);
    }
}
// ✅ Files.walk — stream the tree, filter, collect
try (Stream<Path> paths = Files.walk(Path.of("src"))) {
    List<Path> javaFiles = paths
        .filter(p -> p.toString().endsWith(".java"))
        .toList();
}

// ❌ Switch grows with every new status
double fee(String status, double amount) {
    switch (status) {
        case "NEW": return amount * 0.02;
        case "VIP": return amount * 0.01;
        case "CANCELLED": return 0;
        default: throw new IllegalArgumentException(status);
    }
}
// ✅ Polymorphism — each type owns its fee
interface Priced {
    double fee(double amount);
}
// NewOrder / VipOrder / CancelledOrder each implement Priced
double fee(Priced order, double amount) {
    return order.fee(amount); // no switch to edit
}

java-tips
// ❌ New OS thread per job — costly, unbounded
for (String url : urls) {
    new Thread(() -> http.get(url)).start(); // no limit, hard to join
}
// ✅ Reuse a fixed pool — bounded and manageable
ExecutorService pool = Executors.newFixedThreadPool(8);
try {
    for (String url : urls) {
        pool.submit(() -> http.get(url));
    }
} finally {
    pool.shutdown(); // always stop the pool
}
